```dataviewjs
// ==========================================
// 滚动引用 - 根据标签随机显示语录
// 数据源: [[09 知识库管理/工具/滚动引用/引用数据库]]
// ==========================================

// 0. 清理容器，防止 Obsidian 渲染缓存导致重复显示同一语录
this.container.innerHTML = '';

// 1. 从引用数据库中提取所有语录
// 注意：Dataview 不会解析列表续行中的内联字段，因此从 item.text 中用正则提取
const quotes = dv.pages('"09 知识库管理/工具/滚动引用/引用数据库"')
  .flatMap(p => {
    const results = [];
    const items = p.file.lists;
    for (const item of items) {
      const t = item.text;
      const mc = /类别::\s*(.+?)\s*\n/.exec(t);
      const mq = /语录::\s*(.+?)\s*(\n|$)/.exec(t);
      const ma = /作者::\s*(.+?)\s*(\n|$)/.exec(t);
      if (mc && mq) {
        results.push({
          category: mc[1].trim(),
          text: mq[1].trim(),
          author: (ma && ma[1].trim()) ? ma[1].trim() : null
        });
      }
    }
    return results;
  });

// 2. 获取当前笔记的标签，匹配语录类别
const fileTags = dv.current().tags || [];

// 标签 → 类别 映射关系
const tagToCategory = {
  "交易日志": "交易日志",
  "交易": "交易日志",
  "复盘": "交易日志",
  "月总结": "交易日志",
  "周总结": "交易日志",
  "学习": "学习",
  "学习笔记": "学习",
  "阅读": "学习",
  "专业": "专业",
  "事业": "专业",
  "工作": "专业",
  "项目": "专业",
  "面试": "专业",
  "认知": "认知",
  "思维": "认知",
  "思维模型": "认知",
  "日记": "日记",
  "生活": "日记",
  "健康": "日记",
  "写作": "写作",
  "笔记": "写作",
  "个人成长": "学习"
};

// 3. 查找匹配的语录
let matchedCategory = null;
for (const tag of fileTags) {
  if (tagToCategory[tag]) {
    matchedCategory = tagToCategory[tag];
    break;
  }
}

// 如果标签没匹配到，尝试用路径中的关键词匹配
if (!matchedCategory) {
  const path = dv.current().file.path;
  if (path.contains("交易") || path.contains("执行") || path.contains("财富")) {
    matchedCategory = "交易日志";
  } else if (path.contains("学习") || path.contains("成长")) {
    matchedCategory = "学习";
  } else if (path.contains("专业") || path.contains("事业") || path.contains("项目")) {
    matchedCategory = "专业";
  } else if (path.contains("认知") || path.contains("思维")) {
    matchedCategory = "认知";
  } else if (path.contains("生活") || path.contains("健康") || path.contains("日记")) {
    matchedCategory = "日记";
  } else if (path.contains("写作")) {
    matchedCategory = "写作";
  }
}

const matchedQuotes = matchedCategory
  ? quotes.filter(q => q.category === matchedCategory)
  : [];

// 4. 滚动轮播展示 — 每 5 秒自动切换一条语录
if (matchedQuotes.length > 0) {
  const c = dv.el('div', '');
  const roll = () => {
    const seed = Date.now();
    const r = ((seed * 9301 + 49297) % 233280) / 233280;
    const picked = matchedQuotes[Math.floor(r * matchedQuotes.length)];
    const authorTag = picked.author ? `<br><br>— ${picked.author}` : '';
    c.innerHTML = `<blockquote style="border-left:4px solid var(--blockquote-border-color);padding:0.5em 1em;margin:0">${picked.text}${authorTag}</blockquote>`;
  };
  roll();
  setInterval(roll, 5000);
} else {
  dv.el('blockquote', '所有伟大的交易者都是从记录开始的。继续记录，持续进步。');
}
```
